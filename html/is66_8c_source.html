54void ADC_DMA_write(void)
55{
56 TEST_TP1_Set();
57 /*
58 * trigger both ADC's using PTG trigger
59 */
60 while (AD1STATbits.CH6RDY == 0 && AD2STATbits.CH4RDY == 0); // conversions complete on both
61 adc_result[ADC1_D] = (uint16_t) AD1CH6DATA; // save as uint16_t data
62 adc_result[ADC2_D] = (uint16_t) AD2CH4DATA;
63 memcpy((void *) &iss_adc_write[ADC_SAMPLES_START], (const void *) &adc_result[ADC1_D], ADC_SAMPLES_SIZE);
64 TEST_TP1_Clear();
65
66 switch (iss_state) {
67 case ISS_INIT:
68 store_count = 0;
69 ISS_PAGE_WRITE_CMD = ISS_PAGE_WRITE;
70 iss_page_index = 0;
71 iss_state = ISS_PAGE;
72 iss_page_write_swap = htonl(ISS_PAGE_WRITE_CMD);
73 break;
74 case ISS_PAGE:
75 TEST_TP0_Set();
76 while (SPI2_IsTransmitterBusy() && --cdown != 0); // SPI buffer empty timeout
77 cdown = CDOWN;
78 SRAM_CS_Set();
79 /*
80 * make sure DMA is ready for address sequence
81 */
82 while (DMA_ChannelIsBusy(DMA_CHANNEL_5)) {
83 };
84 SRAM_CS_Clear(); // enable chip

86 * write sram chip data address
87 */
88 DMA_ChannelTransfer(DMA_CHANNEL_5, (const void *) &iss_page_write_swap, (const void*) &SPI2BUF, ISS_WRITE_CMD_SIZE);
89 total_sample_triggers++;
90 if (++iss_page_index >= iss_page_num) {
91 iss_state = ISS_INIT; // back to init state
92 } else {
93 iss_state = ISS_STORE;
94 ISS_PAGE_WRITE_CMD += ISS66_PAGE_SIZE; // next full sram page
95 iss_page_write_swap = htonl(ISS_PAGE_WRITE_CMD);

98 break;
99 case ISS_STORE:
100 while (DMA_ChannelIsBusy(DMA_CHANNEL_5)) {
101 };

104 if (++store_count >= MAX_ISS66_SAMPLES) {
105 store_count = 0;
106 iss_state = ISS_PAGE;
107 }
108 break;
109 case ISS_NULL:
110 break;

169void ADC_DMA_init(void)
170{
171 // Select conversion mode on channel 6 to AN6
172 AD1CH6CONbits.MODE = 3; // 0 = single, 3 = oversample = 2
173 AD2CH4CONbits.MODE = 3;
174 // Set number of conversions accumulated to 2 because of back-to-back.
175 AD1CH6CONbits.ACCNUM = 0;
176 AD2CH4CONbits.ACCNUM = 0;
177 // The oversampling if started cannot be interrupted
178 // by a high priority channels conversion requests.
179 AD1CH6CONbits.ACCBRST = 1;
180 AD2CH4CONbits.ACCBRST = 1;
181 // PTG trigger will start conversions.
182 AD1CH6CONbits.TRG1SRC = 30; // PTG trigger 12
183 AD1CH6CONbits.TRG2SRC = 2; // back-to-back
184 AD2CH4CONbits.TRG1SRC = 30;
185 AD2CH4CONbits.TRG2SRC = 2;
186 // Use a single-ended input.
187 AD1CH6CONbits.DIFF = 0;
188 // Select the AN6 analog positive input/pin for the signal.
189 AD1CH6CONbits.PINSEL = 6;
190 // Select signal sampling time ( 0 = TADs = 6.25nS).
191 AD1CH6CONbits.SAMC = ADC_SAMC;
192 AD2CH4CONbits.SAMC = ADC_SAMC;
193 // Enable repeat rate.
194 AD1CONbits.CALRATE = 1; // Recalibrate every 64 seconds
195 AD2CONbits.CALRATE = 1;
196 // Enable auto calibration.
197 AD1CONbits.ACALEN = 1;
198 AD2CONbits.ACALEN = 1;
199 // Set ADC to RUN mode.
200 AD1CONbits.MODE = 2;
201 // Enable ADC.
202 AD1CONbits.ON = 1;
203 // Wait when ADC will be ready/calibrated.
204 while (AD1CONbits.ADRDY == 0);
205
206 switch (iss_chip) {
207 case ISS_ISS_32Mb:
208 iss_page_num = MAX_ISS66_PAGES_32MBIT;
209 break;
210 case ISS_ISS_16Mb:
211 iss_page_num = MAX_ISS66_PAGES_16MBIT;
212 break;
213 default:
214 iss_page_num = MAX_ISS66_PAGES_8MBIT;
215 break;
216 }
217
218 /*
219 * setup the DMA and timer background tasks
220 */
221 DMA_ChannelCallbackRegister(DMA_CHANNEL_5, SPI2DmaChannelHandler_State, 0);
222 DMA_ChannelCallbackRegister(DMA_CHANNEL_4, SPI2DmaChannelHandler_State_Read, 0);
223 SCCP2_TimerCallbackRegister(SCCP2_Callback_InterruptHandler, (uintptr_t) NULL);
224 SCCP2_Timer32bitPeriodSet(retrigger_time); // close to 2us between sample
225}
