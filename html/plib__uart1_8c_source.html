85static void UART1_ErrorClear( void )
86{
87 UART_ERROR errors = UART_ERROR_NONE;
88 uint8_t dummyData = 0u;
89
90 errors = (UART_ERROR)(U1STAT & (_U1STAT_RXFOIF_MASK | _U1STAT_FERIF_MASK | _U1STAT_PERIF_MASK));
91
92 if(errors != UART_ERROR_NONE)
93 {
94 /* If it's a overrun error then clear it to flush FIFO */
95 if(U1STATbits.RXFOIF != 0U)
96 {
97 U1STATbits.RXFOIF = 0U;
98 }
99
100 /* Read existing error bytes from FIFO to clear parity and framing error flags */
101 while(U1STATbits.RXBE != 1U)
102 {
103 dummyData = (uint8_t)U1RXB;
104 }
105
106 /* Clear error interrupt flag */
107 IFS2bits.U1RXIF = 0;
108
109 /* Clear up the receive interrupt flag so that RX interrupt is not
110 * triggered for error bytes */
111 IFS2bits.U1RXIF = 0;
112
113 }
114
115 // Ignore the warning
116 (void)dummyData;
117}

174bool UART1_SerialSetup( UART_SERIAL_SETUP *setup, uint32_t srcClkFreq )
175{
176 bool status = false;
177 uint32_t baud;
178 uint32_t ctrlReg;
179 uint32_t uxbrg;
180
181 if (setup != NULL)
182 {
183 baud = setup->baudRate;
184
185 if (baud == 0U)
186 {
187 return status;
188 }
189
190
191 srcClkFreq = UART1_FrequencyGet();
192
193
194 /* Turn OFF UART1. Save UTXEN, URXEN bits as these are cleared upon disabling UART */
195 ctrlReg = U1CON & (_U1CON_TXEN_MASK | _U1CON_RXEN_MASK );
196 U1CONbits.ON = 0U;
197
198 /* Calculate BRG value in fractional mode as it has least error rate */
199 uxbrg = (srcClkFreq/baud);
200 /* Check if the valid baud value is set */
201 if(uxbrg < UART_MIN_FRACTIONAL_BAUD)
202 {
203 /* Baud rate cannot be achieved with current clock source value */
204 return status;
205 }
206 else if(uxbrg > UART_MAX_BAUD)
207 {
208 /* Calculate BRG value for high speed mode*/
209 uxbrg = (srcClkFreq/(4U*baud)) - 1U;
210 U1CONbits.BRGS = 1U;
211
212 if(uxbrg > UART_MAX_BAUD)
213 {
214 /* Calculate BRG value for low speed mode*/
215 uxbrg = (srcClkFreq/(16U*baud)) - 1U;
216 U1CONbits.BRGS = 0U;
217 if(uxbrg > UART_MAX_BAUD)
218 {
219 /* Baud rate cannot be achieved with current clock source value */
220 return status;
221 }
222 }
223 }
224 else
225 {
226 U1CONbits.CLKMOD = 1;
227 }
228
229 if(setup->dataWidth == UART_DATA_8_BIT)
230 {
231 /* Configure UART1 mode with parity if mode is 8 bit */
232 U1CONbits.MODE = (uint8_t)setup->parity;
233 }
234 else
235 {
236 /* Configure UART1 mode to 7 bit */
237 U1CONbits.MODE = (uint8_t)setup->dataWidth;
238 }
239
240 /* Configure UART1 mode */
241 U1CONbits.STP = (uint8_t)setup->stopBits;
242
243 /* Configure UART1 Baud Rate */
244 U1BRG = uxbrg;
245
246 uart1Obj.rdBufferSize = UART1_READ_BUFFER_SIZE;
247 uart1Obj.wrBufferSize = UART1_WRITE_BUFFER_SIZE;
248
249 U1CONbits.ON = 1U;
250
251 /* Restore UTXEN, URXEN bits. */
252 U1CON |= ctrlReg;
253
254 status = true;
255 }
256
257 return status;
258}

260/* This routine is only called from ISR. Hence do not disable/enable USART interrupts. */
261static inline bool UART1_RxPushByte(uint8_t rdByte)
262{
263 uint32_t nextIndex;
264 bool isSuccess = false;
265
266 // nextIndex is where head will point to after this write.
267 nextIndex = uart1Obj.rdInIndex + 1U;
268
269 if (nextIndex >= uart1Obj.rdBufferSize)
270 {
271 nextIndex = 0U;
272 }
273
274 if (nextIndex == uart1Obj.rdOutIndex)
275 {
276 /* Queue is full - Report it to the application. Application gets a chance to free up space by reading data out from the RX ring buffer */
277 if(uart1Obj.rdCallback != NULL)
278 {
279 uintptr_t rdContext = uart1Obj.rdContext;
280
281 uart1Obj.rdCallback(UART_EVENT_READ_BUFFER_FULL, rdContext);
282
283 /* Read the indices again in case application has freed up space in RX ring buffer */
284 nextIndex = uart1Obj.rdInIndex + 1U;
285
286 if (nextIndex >= uart1Obj.rdBufferSize)
287 {
288 nextIndex = 0U;
289 }
290 }
291 }
292
293 /* Attempt to push the data into the ring buffer */
294 if (nextIndex != uart1Obj.rdOutIndex)
295 {
296 uint32_t rdInIndex = uart1Obj.rdInIndex;
297
298 UART1_ReadBuffer[rdInIndex] = (uint8_t)rdByte;
299
300 uart1Obj.rdInIndex = nextIndex;
301
302 isSuccess = true;
303 }
304 else
305 {
306 /* Queue is full. Data will be lost. */
307 }
308
309 return isSuccess;
310}

313static void UART1_ReadNotificationSend(void)
314{
315 uint32_t nUnreadBytesAvailable;
316
317 if (uart1Obj.isRdNotificationEnabled == true)
318 {
319 nUnreadBytesAvailable = UART1_ReadCountGet();
320
321 if(uart1Obj.rdCallback != NULL)
322 {
323 uintptr_t rdContext = uart1Obj.rdContext;
324
325 if (uart1Obj.isRdNotifyPersistently == true)
326 {
327 if (nUnreadBytesAvailable >= uart1Obj.rdThreshold)
328 {
329 uart1Obj.rdCallback(UART_EVENT_READ_THRESHOLD_REACHED, rdContext);
330 }
331 }
332 else
333 {
334 if (nUnreadBytesAvailable == uart1Obj.rdThreshold)
335 {
336 uart1Obj.rdCallback(UART_EVENT_READ_THRESHOLD_REACHED, rdContext);
337 }
338 }
339 }

343size_t UART1_Read(uint8_t* pRdBuffer, const size_t size)
344{
345 size_t nBytesRead = 0;
346 uint32_t rdOutIndex = 0;
347 uint32_t rdInIndex = 0;
348
349 /* Take a snapshot of indices to avoid creation of critical section */
350 rdOutIndex = uart1Obj.rdOutIndex;
351 rdInIndex = uart1Obj.rdInIndex;
352
353 while (nBytesRead < size)
354 {
355 if (rdOutIndex != rdInIndex)
356 {
357 pRdBuffer[nBytesRead] = UART1_ReadBuffer[rdOutIndex];
358 nBytesRead++;
359 rdOutIndex++;
360
361 if (rdOutIndex >= uart1Obj.rdBufferSize)
362 {
363 rdOutIndex = 0U;
364 }
365 }
366 else
367 {
368 /* No more data available in the RX buffer */
369 break;
370 }
371 }
372
373 uart1Obj.rdOutIndex = rdOutIndex;
374
375 return nBytesRead;
376}

552size_t UART1_Write(uint8_t* pWrBuffer, const size_t size )
553{
554 size_t nBytesWritten = 0U;
555
556 while (nBytesWritten < size)
557 {
558
559 if (UART1_TxPushByte(pWrBuffer[nBytesWritten]) == true)
560 {
561 nBytesWritten++;
562 }
563 else
564 {
565 /* Queue is full, exit the loop */
566 break;
567 }
568
569 }
570
571 /* Check if any data is pending for transmission */
572 if (UART1_WritePendingBytesGet() > 0U)
573 {
574 /* Enable TX interrupt as data is pending for transmission */
575 IEC2bits.U1TXIE = 1U;
576 }
577
578 return nBytesWritten;
579}

685void U1RX_InterruptHandler(void)
686{
687 /* Keep reading until there is a character available in the RX FIFO */
688 while(U1STATbits.RXBE != 1U)
689 {
690 if (UART1_RxPushByte((uint8_t )(U1RXB)) == true)
691 {
692 UART1_ReadNotificationSend();
693 }
694 else
695 {
696 /* UART RX buffer is full */
697 }
698 }
699
700 /* Clear UART1 RX Interrupt flag */
701 IFS2bits.U1RXIF = 0U;

704void U1TX_InterruptHandler(void)
705{
706 uint16_t wrByte;
707
708 /* Check if any data is pending for transmission */
709 if (UART1_WritePendingBytesGet() > 0U)
710 {
711 /* Clear UART1TX Interrupt flag */
712 IFS2bits.U1TXIF = 0U;
713
714 /* Keep writing to the TX FIFO as long as there is space */
715 while(U1STATbits.TXBF == 0U)
716 {
717 if (UART1_TxPullByte(&wrByte) == true)
718 {
719 U1TXB = (uint8_t)wrByte;
720
721 /* Send notification */
722 UART1_WriteNotificationSend();
723 }
724 else
725 {
726 /* Nothing to transmit. Disable the data register empty interrupt. */
727 IEC2bits.U1TXIE = 0U;
728 break;
729 }
730 }
731 }
732 else
733 {
734 /* Nothing to transmit. Disable the data register empty interrupt. */
735 IEC2bits.U1TXIE = 0U;
736 }
