49bool Quaternion_equal(Quaternion* q1, Quaternion* q2)
50{
51 bool equalW = fabs(q1->w - q2->w) <= QUATERNION_EPS;
52 bool equalV0 = fabs(q1->v[0] - q2->v[0]) <= QUATERNION_EPS;
53 bool equalV1 = fabs(q1->v[1] - q2->v[1]) <= QUATERNION_EPS;
54 bool equalV2 = fabs(q1->v[2] - q2->v[2]) <= QUATERNION_EPS;
55 return equalW && equalV0 && equalV1 && equalV2;
56}

65void Quaternion_fromAxisAngle(double axis[3], double angle, Quaternion* output)
66{
67 assert(output != NULL);
68 // Formula from http://www.euclideanspace.com/maths/geometry/rotations/conversions/angleToQuaternion/
69 output->w = cos(angle / 2.0);
70 double c = sin(angle / 2.0);
71 output->v[0] = c * axis[0];
72 output->v[1] = c * axis[1];
73 output->v[2] = c * axis[2];
74}

76double Quaternion_toAxisAngle(Quaternion* q, double output[3])
77{
78 assert(output != NULL);
79 // Formula from http://www.euclideanspace.com/maths/geometry/rotations/conversions/quaternionToAngle/
80 double angle = 2.0 * acos(q->w);
81 double divider = sqrt(1.0 - q->w * q->w);
82
83 if(divider != 0.0) {
84 // Calculate the axis
85 output[0] = q->v[0] / divider;
86 output[1] = q->v[1] / divider;
87 output[2] = q->v[2] / divider;
88 } else {
89 // Arbitrary normalized axis
90 output[0] = 1;
91 output[1] = 0;
92 output[2] = 0;
93 }
94 return angle;
95}

118void Quaternion_fromEulerZYX(double eulerZYX[3], Quaternion* output)
119{
120 assert(output != NULL);
121 // Based on https://en.wikipedia.org/wiki/Conversion_between_quaternions_and_Euler_angles
122 double cy = cos(eulerZYX[2] * 0.5);
123 double sy = sin(eulerZYX[2] * 0.5);
124 double cr = cos(eulerZYX[0] * 0.5);
125 double sr = sin(eulerZYX[0] * 0.5);
126 double cp = cos(eulerZYX[1] * 0.5);
127 double sp = sin(eulerZYX[1] * 0.5);
128
129 output->w = cy * cr * cp + sy * sr * sp;
130 output->v[0] = cy * sr * cp - sy * cr * sp;
131 output->v[1] = cy * cr * sp + sy * sr * cp;
132 output->v[2] = sy * cr * cp - cy * sr * sp;
133}

135void Quaternion_toEulerZYX(Quaternion* q, double output[3])
136{
137 assert(output != NULL);
138
139 // Roll (x-axis rotation)
140 double sinr_cosp = +2.0 * (q->w * q->v[0] + q->v[1] * q->v[2]);
141 double cosr_cosp = +1.0 - 2.0 * (q->v[0] * q->v[0] + q->v[1] * q->v[1]);
142 output[0] = atan2(sinr_cosp, cosr_cosp);
143
144 // Pitch (y-axis rotation)
145 double sinp = +2.0 * (q->w * q->v[1] - q->v[2] * q->v[0]);
146 if (fabs(sinp) >= 1)
147 output[1] = copysign(M_PI / 2, sinp); // use 90 degrees if out of range
148 else
149 output[1] = asin(sinp);
150
151 // Yaw (z-axis rotation)
152 double siny_cosp = +2.0 * (q->w * q->v[2] + q->v[0] * q->v[1]);
153 double cosy_cosp = +1.0 - 2.0 * (q->v[1] * q->v[1] + q->v[2] * q->v[2]);
154 output[2] = atan2(siny_cosp, cosy_cosp);
155}

184void Quaternion_multiply(Quaternion* q1, Quaternion* q2, Quaternion* output)
185{
186 assert(output != NULL);
187 Quaternion result;
188
189 /*
190 Formula from http://www.euclideanspace.com/maths/algebra/realNormedAlgebra/quaternions/arithmetic/index.htm
191 a*e - b*f - c*g - d*h
192 + i (b*e + a*f + c*h- d*g)
193 + j (a*g - b*h + c*e + d*f)
194 + k (a*h + b*g - c*f + d*e)
195 */
196 result.w = q1->w *q2->w - q1->v[0]*q2->v[0] - q1->v[1]*q2->v[1] - q1->v[2]*q2->v[2];
197 result.v[0] = q1->v[0]*q2->w + q1->w *q2->v[0] + q1->v[1]*q2->v[2] - q1->v[2]*q2->v[1];
198 result.v[1] = q1->w *q2->v[1] - q1->v[0]*q2->v[2] + q1->v[1]*q2->w + q1->v[2]*q2->v[0];
199 result.v[2] = q1->w *q2->v[2] + q1->v[0]*q2->v[1] - q1->v[1]*q2->v[0] + q1->v[2]*q2->w ;
200
201 *output = result;
202}

204void Quaternion_rotate(Quaternion* q, double v[3], double output[3])
205{
206 assert(output != NULL);
207 double result[3];
208
209 double ww = q->w * q->w;
210 double xx = q->v[0] * q->v[0];
211 double yy = q->v[1] * q->v[1];
212 double zz = q->v[2] * q->v[2];
213 double wx = q->w * q->v[0];
214 double wy = q->w * q->v[1];
215 double wz = q->w * q->v[2];
216 double xy = q->v[0] * q->v[1];
217 double xz = q->v[0] * q->v[2];
218 double yz = q->v[1] * q->v[2];
219
220 // Formula from http://www.euclideanspace.com/maths/algebra/realNormedAlgebra/quaternions/transforms/index.htm
221 // p2.x = w*w*p1.x + 2*y*w*p1.z - 2*z*w*p1.y + x*x*p1.x + 2*y*x*p1.y + 2*z*x*p1.z - z*z*p1.x - y*y*p1.x;
222 // p2.y = 2*x*y*p1.x + y*y*p1.y + 2*z*y*p1.z + 2*w*z*p1.x - z*z*p1.y + w*w*p1.y - 2*x*w*p1.z - x*x*p1.y;
223 // p2.z = 2*x*z*p1.x + 2*y*z*p1.y + z*z*p1.z - 2*w*y*p1.x - y*y*p1.z + 2*w*x*p1.y - x*x*p1.z + w*w*p1.z;
224
225 result[0] = ww*v[0] + 2*wy*v[2] - 2*wz*v[1] +
226 xx*v[0] + 2*xy*v[1] + 2*xz*v[2] -
227 zz*v[0] - yy*v[0];
228 result[1] = 2*xy*v[0] + yy*v[1] + 2*yz*v[2] +
229 2*wz*v[0] - zz*v[1] + ww*v[1] -
230 2*wx*v[2] - xx*v[1];
231 result[2] = 2*xz*v[0] + 2*yz*v[1] + zz*v[2] -
232 2*wy*v[0] - yy*v[2] + 2*wx*v[1] -
233 xx*v[2] + ww*v[2];
234
235 // Copy result to output
236 output[0] = result[0];
237 output[1] = result[1];
238 output[2] = result[2];
239}

241void Quaternion_slerp(Quaternion* q1, Quaternion* q2, double t, Quaternion* output)
242{
243 Quaternion result;
244
245 // Based on http://www.euclideanspace.com/maths/algebra/realNormedAlgebra/quaternions/slerp/index.htm
246 double cosHalfTheta = q1->w*q2->w + q1->v[0]*q2->v[0] + q1->v[1]*q2->v[1] + q1->v[2]*q2->v[2];
247
248 // if q1=q2 or qa=-q2 then theta = 0 and we can return qa
249 if (fabs(cosHalfTheta) >= 1.0) {
250 Quaternion_copy(q1, output);
251 return;
252 }
253
254 double halfTheta = acos(cosHalfTheta);
255 double sinHalfTheta = sqrt(1.0 - cosHalfTheta*cosHalfTheta);
256 // If theta = 180 degrees then result is not fully defined
257 // We could rotate around any axis normal to q1 or q2
258 if (fabs(sinHalfTheta) < QUATERNION_EPS) {
259 result.w = (q1->w * 0.5 + q2->w * 0.5);
260 result.v[0] = (q1->v[0] * 0.5 + q2->v[0] * 0.5);
261 result.v[1] = (q1->v[1] * 0.5 + q2->v[1] * 0.5);
262 result.v[2] = (q1->v[2] * 0.5 + q2->v[2] * 0.5);
263 } else {
264 // Default quaternion calculation
265 double ratioA = sin((1 - t) * halfTheta) / sinHalfTheta;
266 double ratioB = sin(t * halfTheta) / sinHalfTheta;
267 result.w = (q1->w * ratioA + q2->w * ratioB);
268 result.v[0] = (q1->v[0] * ratioA + q2->v[0] * ratioB);
269 result.v[1] = (q1->v[1] * ratioA + q2->v[1] * ratioB);
270 result.v[2] = (q1->v[2] * ratioA + q2->v[2] * ratioB);
271 }
272 *output = result;
273}
