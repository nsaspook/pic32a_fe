102void CMP1_Initialize(void)
103{
104 // Comparator Register settings
105 DACCTRL1bits.FCLKDIV = 0U;
106 DAC1CON = (DAC1CON_TMCB_SET(0x0U)
107 |DAC1CON_IRQM_DISABLED
108 |_DAC1CON_DACOEN_MASK
109 |DAC1CON_INSEL_CMPxA
110 |DAC1CON_HYSSEL_NO_HYS);
111
112 //DAC Settings
113 DAC1DAT = (DAC1DAT_DACDAT_SET(0xf32UL)
114 |DAC1DAT_DACLOW_SET(0xcdUL));
115
116 DAC1SLPCON = (DAC1SLPCON_HCFSEL_SET(0)
117 |DAC1SLPCON_SLPSTOPA_SET(0)
118 |DAC1SLPCON_SLPSTOPB_SET(0)
119 |DAC1SLPCON_SLPSTRT_SET(0));
120
121 DAC1SLPDAT = 0x0UL;
122
123 DAC1CONbits.DACEN = 1U;
124 DACCTRL1bits.ON = 1U;
125}

127void CMP1_Deinitialize(void)

143void CMP1_Calibrate(void)
144{
145 uint32_t *fpdmdac = (uint32_t*)CMP_FPDMDAC_ADDRESS;
146 uint32_t fpdmdac_value = *fpdmdac;
147
148 DACCTRL1bits.POSINLADJ = (uint8_t)((fpdmdac_value & _DACCTRL1_POSINLADJ_MASK) >> _DACCTRL1_POSINLADJ_POSITION);
149 DACCTRL1bits.NEGINLADJ = (uint8_t)((fpdmdac_value & _DACCTRL1_NEGINLADJ_MASK) >> _DACCTRL1_NEGINLADJ_POSITION);
150 DACCTRL1bits.DNLADJ = (uint8_t)((fpdmdac_value & _DACCTRL1_DNLADJ_MASK) >> _DACCTRL1_DNLADJ_POSITION);
151 DACCTRL1bits.RREN = 1U;
152}
