65void TMR1_Initialize(void)
66{
67 /* Disable Timer */
68 T1CONbits.ON = 0;
69
70 T1CON = 0x0UL;
71 /* Clear counter */
72 TMR1 = 0x0UL;
73
74 /*Set period */
75 PR1 = 0xc34fUL; /* Decimal Equivalent 49999 */
76
77 tmr1Obj.tickCounter = 0;
78 tmr1Obj.callback_fn = NULL;
79
80 /* Setup TMR1 Interrupt */
81 TMR1_InterruptEnable(); /* Enable interrupt on the way out */
82}
83
84void TMR1_Deinitialize(void)
85{
86 /* Stopping the timer */
87 TMR1_Stop();
88
89 /* Deinitializing the registers to POR values */
90 T1CON = 0x0UL;
91 TMR1 = 0x0UL;
92 PR1 = 0xFFFFFFFFUL;
93}

145{

156void __attribute__((used)) T1_InterruptHandler (void)
157{
158 uint32_t status = _T1IF;
159 _T1IF = 0;
160
161 tmr1Obj.tickCounter++;
162
163 if((tmr1Obj.callback_fn != NULL))
164 {
165 uintptr_t context = tmr1Obj.context;
166 tmr1Obj.callback_fn(status, context);
167 }
168}
