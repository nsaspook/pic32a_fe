16// Calculate CRC for 24 MSB's of the 32 bit dword
17// (8 LSB's are the CRC field and are not included in CRC calculation)
18
19uint8_t CalculateCRC(uint32_t Data)
20{
21 uint8_t BitIndex;
22 uint8_t BitValue;
23 uint8_t CRC;
24
25 CRC = 0xFF;
26 for (BitIndex = 31; BitIndex > 7; BitIndex--) {
27 BitValue = (uint8_t) ((Data >> BitIndex) & 0x01);
28 CRC = CRC8(BitValue, CRC);
29 }
30 CRC = (uint8_t) ~CRC;
31 return CRC;
32}
33
34static uint8_t CRC8(uint8_t BitValue, uint8_t CRC)
35{
36 uint8_t Temp;
37
38 Temp = (uint8_t) (CRC & 0x80);
39 if (BitValue == 0x01) {
40 Temp ^= 0x80;
41 }
42 CRC <<= 1;
43 if (Temp > 0) {
44 CRC ^= 0x1D;
45 }
46 return CRC;
47}
