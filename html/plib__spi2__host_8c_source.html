87void SPI2_Initialize ( void )
88{
89 /* Disable SPI2 Interrupts */
90 IEC2bits.SPI2EIE = 0U;
91 IEC2bits.SPI2RXIE = 0U;
92 IEC2bits.SPI2TXIE = 0U;
93
94 /* STOP and Reset the SPI */
95 SPI2CON1 = 0x00UL;
96
97 /* Clear SPI2 Interrupt flags */
98 IFS2bits.SPI2EIF = 0U;
99 IFS2bits.SPI2RXIF = 0U;
100 IFS2bits.SPI2TXIF = 0U;
101
102 /* BAUD Rate register Setup */
103 SPI2BRG = 0x0UL;
104
105 SPI2CON1 = (SPI2CON1_MSTEN_HOST_MODE
106 |SPI2CON1_CKP_IDLE_LOW_ACTIVE_HIGH
107 |SPI2CON1_CKE_ACTIVE_TO_IDLE
108 |SPI2CON1_MCLKEN_CLOCK_GEN_9
109 |_SPI2CON1_SMP_MASK
110 |_SPI2CON1_ENHBUF_MASK);
111
112 /* Initialize global variables */
113 spi2Obj.transferIsBusy = false;
114 spi2Obj.callback = NULL;
115
116 /* Enable SPI2 */
117 SPI2CON1bits.ON = 1U;
118}

145bool SPI2_TransferSetup (SPI_TRANSFER_SETUP* setup, uint32_t spiSourceClock )
146{
147 uint32_t t_brg;
148 uint32_t baudHigh;
149 uint32_t baudLow;
150 uint32_t errorHigh;
151 uint32_t errorLow;
152
153 if ((setup == NULL) || (setup->clockFrequency == 0U) || (setup->dataBits != SPI_DATA_BITS_8))
154 {
155 return false;
156 }
157
158 spiSourceClock = SPI2_FrequencyGet();
159
160 t_brg = (((spiSourceClock / (setup->clockFrequency)) / 2u) - 1u);
161 baudHigh = spiSourceClock / (2u * (t_brg + 1u));
162 baudLow = spiSourceClock / (2u * (t_brg + 2u));
163 errorHigh = baudHigh - setup->clockFrequency;
164 errorLow = setup->clockFrequency - baudLow;
165
166 if (errorHigh > errorLow)
167 {
168 t_brg++;
169 }
170
171 if(t_brg > 0x1fffUL)
172 {
173 return false;
174 }
175
176 /* STOP and Reset the SPI */
177 SPI2CON1bits.ON = 0U;
178
179 SPI2BRG = t_brg;
180
181 SPI2CON1 = (SPI2CON1 & (~(_SPI2CON1_MODE16_MASK | _SPI2CON1_MODE32_MASK | _SPI2CON1_CKP_MASK | _SPI2CON1_CKE_MASK))) |
182 ((uint32_t)setup->clockPolarity | (uint32_t)setup->clockPhase | (uint32_t)setup->dataBits);
183
184 SPI2CON1bits.ON = 1U;
185
186 return true;
187}

218static void SPI_FIFO_Fill(void)
219{
220 uint8_t nDataCopiedToFIFO = 0U;
221
222 size_t txCount = spi2Obj.txCount;
223
224 while (nDataCopiedToFIFO < SPI2_FIFO_SIZE)
225 {
226 if(SPI2STATbits.SPITBF != 0U)
227 {
228 break; /* Exit loop if buffer is full */
229 }
230 if (txCount < spi2Obj.txSize)
231 {
232 SPI2BUF = ((uint8_t*)spi2Obj.txBuffer)[txCount];
233 txCount++;
234 }
235 else if (spi2Obj.dummySize > 0U)
236 {
237 SPI2BUF = SPI2_DUMMY_DATA;
238 spi2Obj.dummySize--;
239 }
240 else
241 {
242 break;
243 }
244 nDataCopiedToFIFO++;
245 }
246 spi2Obj.txCount = txCount;
247}

249bool SPI2_WriteRead (void* pTransmitData, size_t txSize, void* pReceiveData, size_t rxSize)
250{
251 bool isRequestAccepted = false;
252
253 /* Verify the request */
254 if((spi2Obj.transferIsBusy == false) && (((txSize > 0U) && (pTransmitData != NULL)) || ((rxSize > 0U) && (pReceiveData != NULL))))
255 {
256 isRequestAccepted = true;
257 spi2Obj.txBuffer = pTransmitData;
258 spi2Obj.rxBuffer = pReceiveData;
259 spi2Obj.rxCount = 0U;
260 spi2Obj.txCount = 0U;
261 spi2Obj.dummySize = 0U;
262
263 if (pTransmitData != NULL)
264 {
265 spi2Obj.txSize = txSize;
266 }
267 else
268 {
269 spi2Obj.txSize = 0U;
270 }
271
272 if (pReceiveData != NULL)
273 {
274 spi2Obj.rxSize = rxSize;
275 }
276 else
277 {
278 spi2Obj.rxSize = 0U;
279 }
280
281 spi2Obj.transferIsBusy = true;
282
283 size_t txSz = spi2Obj.txSize;
284 if (spi2Obj.rxSize > txSz)
285 {
286 spi2Obj.dummySize = spi2Obj.rxSize - txSz;
287 }
288
289 /* Clear the receive overflow error if any */
290 SPI2STATbits.SPIROV = 0U;
291
292 /* Make sure there is no data pending in the RX FIFO */
293
294 while ((SPI2STAT & _SPI2STAT_SPIRBE_MASK) == 0U)
295 {
296 (void)SPI2BUF;
297 }
298 /* Configure SPI to generate receive interrupt when receive buffer is empty */
299 SPI2IMSKbits.RXMSK = 1U;
300 SPI2IMSKbits.RXWIEN = 1U;
301
302 /* Configure SPI to generate transmit interrupt when the transmit buffer is empty*/
303 SPI2IMSKbits.SPITBEN = 1U;
304
305 /* Disable the receive interrupt */
306 IEC2bits.SPI2RXIE = 0U;
307
308 /* Disable the transmit interrupt */
309 IEC2bits.SPI2TXIE = 0U;
310
311 /* Disable the error interrupt */
312 IEC2bits.SPI2EIE = 0U;
313
314 /* Clear the receive interrupt flag */
315 IFS2bits.SPI2RXIF = 0U;
316
317 /* Clear the transmit interrupt flag */
318 IFS2bits.SPI2TXIF = 0U;
319
320 /* Clear the error interrupt flag */
321 IFS2bits.SPI2EIF = 0U;
322
323 /* Start the first write here itself, rest will happen in ISR context */
324 SPI_FIFO_Fill();
325
326 if (rxSize > 0U)
327 {
328 if(rxSize < SPI2_FIFO_SIZE)
329 {
330 SPI2IMSKbits.RXMSK = (uint8_t)rxSize;
331 }
332 else
333 {
334 SPI2IMSKbits.RXMSK = SPI2_FIFO_SIZE;
335 }
336 /* Enable receive interrupt to complete the transfer in ISR context.
337 * Keep the transmit interrupt disabled. Transmit interrupt will be
338 * enabled later if txCount < txSize, when rxCount = rxSize.
339 */
340 IEC2bits.SPI2RXIE = 1U;
341 }
342 else
343 {
344 if (spi2Obj.txCount != txSz)
345 {
346 /* Configure SPI to generate interrupt when transmit buffer is completely empty */
347 SPI2IMSKbits.SPITBEN = 1U;
348
349 /* ignore receive overflow for transmit only operation*/
350 SPI2CON1bits.IGNROV = 1U;
351
352 /* Enable transmit interrupt to complete the transfer in ISR context */
353 IEC2bits.SPI2TXIE = 1U;
354
355 }
356 else
357 {
358 /* Enable error interrupt for SRMT(last byte transfer in shift register)*/
359 SPI2IMSKbits.SRMTEN = 1U;
360 IEC2bits.SPI2EIE = 1U;
361 }
362
363 }
364 }
365
366 return isRequestAccepted;
367}

369void __attribute__((used)) SPI2RX_InterruptHandler (void)
370{
371 uint32_t nRxPending = 0;
372 uint32_t receivedData = 0;
373
374 /* Check Receive Buffer Element Count for watermark interrupt */
375 if ((SPI2STAT & _SPI2STAT_RXELM_MASK) != 0U)
376 {
377 size_t txCount = spi2Obj.txCount;
378 size_t rxCount = spi2Obj.rxCount;
379 while(SPI2STATbits.SPIRBE == false)
380 {
381 if (rxCount < spi2Obj.rxSize)
382 {
383 /* Receive buffer is not empty. Read the received data. */
384 receivedData = SPI2BUF;
385
386 ((uint8_t*)spi2Obj.rxBuffer)[rxCount] = (uint8_t)receivedData;
387 rxCount++;
388
389 spi2Obj.rxCount = rxCount;
390
391 if (rxCount == spi2Obj.rxSize)
392 {
393 if (txCount < spi2Obj.txSize)
394 {
395 /* Reception of all bytes is complete. However, there are few more
396 * bytes to be transmitted as txCount != txSize. Finish the
397 * transmission of the remaining bytes from the transmit interrupt. */
398
399 /* Disable the receive interrupt */
400 IEC2bits.SPI2RXIE = 0U;
401
402 /* Generate TX interrupt when buffer is completely empty */
403 SPI2IMSKbits.SPITBEN = 1U;
404
405 /* Enable the transmit interrupt. Callback will be given from the
406 * transmit interrupt, when all bytes are shifted out. */
407 IEC2bits.SPI2TXIE = 1U;
408 }
409 }
410 }
411 }
412 if (rxCount < spi2Obj.rxSize)
413 {
414 /* More bytes pending to be received .. */
415 SPI_FIFO_Fill();
416
417 nRxPending = spi2Obj.rxSize - rxCount;
418
419 if(nRxPending <= SPI2_FIFO_SIZE)
420 {
421 SPI2IMSKbits.RXMSK = (uint8_t)nRxPending;
422 }
423 else
424 {
425 SPI2IMSKbits.RXMSK = SPI2_FIFO_SIZE;
426 }
427 }
428 else
429 {
430 if(rxCount == spi2Obj.rxSize)
431 {
432 if (txCount == spi2Obj.txSize)
433 {
434 /* Clear receiver overflow error if any */
435 SPI2STATbits.SPIROV = 0U;
436
437 /* Disable receive interrupt */
438 IEC2bits.SPI2RXIE = 0U;
439
440 /* Transfer complete. Give a callback */
441 spi2Obj.transferIsBusy = false;
442
443 if(spi2Obj.callback != NULL)
444 {
445 uintptr_t context = spi2Obj.context;
446 spi2Obj.callback(context);
447 }
448 }
449 }
450 }
451 }
452
453 /* Clear SPI2 RX Interrupt flag */
454 /* This flag should cleared only after reading buffer */
455 IFS2bits.SPI2RXIF = 0U;
456}

485void __attribute__((used)) SPI2TX_InterruptHandler (void)
486{
487 /* If there are more words to be transmitted, then transmit them here and keep track of the count */
488 if((SPI2STAT & _SPI2STAT_SPITBE_MASK) == _SPI2STAT_SPITBE_MASK)
489 {
490 size_t txCount = spi2Obj.txCount;
491
492 while (txCount < spi2Obj.txSize)
493 {
494 if(SPI2STATbits.SPITBF != 0U)
495 {
496 break; /* Exit loop if buffer is full */
497 }
498 SPI2BUF = ((uint8_t*)spi2Obj.txBuffer)[txCount];
499 txCount++;
500
501 spi2Obj.txCount = txCount;
502 if (txCount == spi2Obj.txSize)
503 {
504 /* All bytes are submitted to the SPI module. Now, enable transmit
505 * interrupt for the shift register to empty */
506 SPI2IMSKbits.SPITBEN = 0U;
507 SPI2IMSKbits.SRMTEN = 1U;
508
509 /* Enable the error interrupt and disable the transmit interrupt*/
510 IEC2bits.SPI2EIE = 1U;
511 IEC2bits.SPI2TXIE = 0U;
512 }
513
514 }
515 }
516 /* Clear the transmit interrupt flag */
517 IFS2bits.SPI2TXIF = 0U;
518}
