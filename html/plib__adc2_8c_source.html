120void ADC2_Initialize(void)
121{
122 AD2CON = AD2CON_RPTCNT_SET(18);
123 //Clear CMP status
124 AD2CMPSTAT = 0x0U;
125
126 AD2CH4CON = (AD2CH4CON_MODE_SINGLE_SAMPLE
127 |AD2CH4CON_ACCNUM_4_SAMPLES
128 |AD2CH4CON_TRG2SRC_SET(0)
129 |AD2CH4CON_CMPMOD_DISABLED
130 |AD2CH4CON_PINSEL_SET(7)
131 |AD2CH4CON_NINSEL_SET(0)
132 |AD2CH4CON_SAMC_SET(0)
133 |AD2CH4CON_TRG1SRC_SET(30));
134 AD2CH4CNT = AD2CH4CNT_CNT_SET(0);
135 AD2CH4CMPLO = 0X0UL;
136 AD2CH4CMPHI = 0X0UL;
137
138
139 //Mode change to run mode and enable ADC
140 AD2CON |= (AD2CON_MODE_ON | _AD2CON_ON_MASK);
141 while(AD2CONbits.ADRDY == 0U){};
142}

168static uint16_t ADC2_TriggerSourceValueGet(ADC2_PWM_INSTANCE pwmInstance, ADC_PWM_TRIGGERS triggerNumber)
169{
170 uint16_t adcTriggerSourceValue = 0x0U;
171 switch(pwmInstance)
172 {
173 case ADC2_PWM4:
174 if(triggerNumber == ADC_PWM_TRIGGER_1)
175 {
176 adcTriggerSourceValue = (uint16_t)PWM4_TRIGGER1;
177 }
178 else if(triggerNumber == ADC_PWM_TRIGGER_2)
179 {
180 adcTriggerSourceValue = (uint16_t)PWM4_TRIGGER2;
181 }
182 else
183 {
184 /*Do Nothing*/
185 }
186 break;
187 default:
188 /*Do Nothing*/
189 break;
190 }
191 return adcTriggerSourceValue;
192}
