87void SPI1_Initialize ( void )
88{
89 /* Disable SPI1 Interrupts */
90 IEC2bits.SPI1EIE = 0U;
91 IEC1bits.SPI1RXIE = 0U;
92 IEC2bits.SPI1TXIE = 0U;
93
94 /* STOP and Reset the SPI */
95 SPI1CON1 = 0x00UL;
96
97 /* Clear SPI1 Interrupt flags */
98 IFS2bits.SPI1EIF = 0U;
99 IFS1bits.SPI1RXIF = 0U;
100 IFS2bits.SPI1TXIF = 0U;
101
102 /* BAUD Rate register Setup */
103 SPI1BRG = 0x1UL;
104
105 SPI1CON1 = (SPI1CON1_MSTEN_HOST_MODE
106 |SPI1CON1_CKP_IDLE_HIGH_ACTIVE_LOW
107 |SPI1CON1_CKE_IDLE_TO_ACTIVE
108 |SPI1CON1_MCLKEN_CLOCK_GEN_9
109 |_SPI1CON1_ENHBUF_MASK);
110
111 /* Initialize global variables */
112 spi1Obj.transferIsBusy = false;
113 spi1Obj.callback = NULL;
114
115 /* Enable SPI1 */
116 SPI1CON1bits.ON = 1U;
117}

144bool SPI1_TransferSetup (SPI_TRANSFER_SETUP* setup, uint32_t spiSourceClock )
145{
146 uint32_t t_brg;
147 uint32_t baudHigh;
148 uint32_t baudLow;
149 uint32_t errorHigh;
150 uint32_t errorLow;
151
152 if ((setup == NULL) || (setup->clockFrequency == 0U) || (setup->dataBits != SPI_DATA_BITS_8))
153 {
154 return false;
155 }
156
157 spiSourceClock = SPI1_FrequencyGet();
158
159 t_brg = (((spiSourceClock / (setup->clockFrequency)) / 2u) - 1u);
160 baudHigh = spiSourceClock / (2u * (t_brg + 1u));
161 baudLow = spiSourceClock / (2u * (t_brg + 2u));
162 errorHigh = baudHigh - setup->clockFrequency;
163 errorLow = setup->clockFrequency - baudLow;
164
165 if (errorHigh > errorLow)
166 {
167 t_brg++;
168 }
169
170 if(t_brg > 0x1fffUL)
171 {
172 return false;
173 }
174
175 /* STOP and Reset the SPI */
176 SPI1CON1bits.ON = 0U;
177
178 SPI1BRG = t_brg;
179
180 SPI1CON1 = (SPI1CON1 & (~(_SPI1CON1_MODE16_MASK | _SPI1CON1_MODE32_MASK | _SPI1CON1_CKP_MASK | _SPI1CON1_CKE_MASK))) |
181 ((uint32_t)setup->clockPolarity | (uint32_t)setup->clockPhase | (uint32_t)setup->dataBits);
182
183 SPI1CON1bits.ON = 1U;
184
185 return true;
186}

217static void SPI_FIFO_Fill(void)
218{
219 uint8_t nDataCopiedToFIFO = 0U;
220
221 size_t txCount = spi1Obj.txCount;
222
223 while (nDataCopiedToFIFO < SPI1_FIFO_SIZE)
224 {
225 if(SPI1STATbits.SPITBF != 0U)
226 {
227 break; /* Exit loop if buffer is full */
228 }
229 if (txCount < spi1Obj.txSize)
230 {
231 SPI1BUF = ((uint8_t*)spi1Obj.txBuffer)[txCount];
232 txCount++;
233 }
234 else if (spi1Obj.dummySize > 0U)
235 {
236 SPI1BUF = SPI1_DUMMY_DATA;
237 spi1Obj.dummySize--;
238 }
239 else
240 {
241 break;
242 }
243 nDataCopiedToFIFO++;
244 }
245 spi1Obj.txCount = txCount;
246}

248bool SPI1_WriteRead (void* pTransmitData, size_t txSize, void* pReceiveData, size_t rxSize)
249{
250 bool isRequestAccepted = false;
251
252 /* Verify the request */
253 if((spi1Obj.transferIsBusy == false) && (((txSize > 0U) && (pTransmitData != NULL)) || ((rxSize > 0U) && (pReceiveData != NULL))))
254 {
255 isRequestAccepted = true;
256 spi1Obj.txBuffer = pTransmitData;
257 spi1Obj.rxBuffer = pReceiveData;
258 spi1Obj.rxCount = 0U;
259 spi1Obj.txCount = 0U;
260 spi1Obj.dummySize = 0U;
261
262 if (pTransmitData != NULL)
263 {
264 spi1Obj.txSize = txSize;
265 }
266 else
267 {
268 spi1Obj.txSize = 0U;
269 }
270
271 if (pReceiveData != NULL)
272 {
273 spi1Obj.rxSize = rxSize;
274 }
275 else
276 {
277 spi1Obj.rxSize = 0U;
278 }
279
280 spi1Obj.transferIsBusy = true;
281
282 size_t txSz = spi1Obj.txSize;
283 if (spi1Obj.rxSize > txSz)
284 {
285 spi1Obj.dummySize = spi1Obj.rxSize - txSz;
286 }
287
288 /* Clear the receive overflow error if any */
289 SPI1STATbits.SPIROV = 0U;
290
291 /* Make sure there is no data pending in the RX FIFO */
292
293 while ((SPI1STAT & _SPI1STAT_SPIRBE_MASK) == 0U)
294 {
295 (void)SPI1BUF;
296 }
297 /* Configure SPI to generate receive interrupt when receive buffer is empty */
298 SPI1IMSKbits.RXMSK = 1U;
299 SPI1IMSKbits.RXWIEN = 1U;
300
301 /* Configure SPI to generate transmit interrupt when the transmit buffer is empty*/
302 SPI1IMSKbits.SPITBEN = 1U;
303
304 /* Disable the receive interrupt */
305 IEC1bits.SPI1RXIE = 0U;
306
307 /* Disable the transmit interrupt */
308 IEC2bits.SPI1TXIE = 0U;
309
310 /* Disable the error interrupt */
311 IEC2bits.SPI1EIE = 0U;
312
313 /* Clear the receive interrupt flag */
314 IFS1bits.SPI1RXIF = 0U;
315
316 /* Clear the transmit interrupt flag */
317 IFS2bits.SPI1TXIF = 0U;
318
319 /* Clear the error interrupt flag */
320 IFS2bits.SPI1EIF = 0U;
321
322 /* Start the first write here itself, rest will happen in ISR context */
323 SPI_FIFO_Fill();
324
325 if (rxSize > 0U)
326 {
327 if(rxSize < SPI1_FIFO_SIZE)
328 {
329 SPI1IMSKbits.RXMSK = (uint8_t)rxSize;
330 }
331 else
332 {
333 SPI1IMSKbits.RXMSK = SPI1_FIFO_SIZE;
334 }
335 /* Enable receive interrupt to complete the transfer in ISR context.
336 * Keep the transmit interrupt disabled. Transmit interrupt will be
337 * enabled later if txCount < txSize, when rxCount = rxSize.
338 */
339 IEC1bits.SPI1RXIE = 1U;
340 }
341 else
342 {
343 if (spi1Obj.txCount != txSz)
344 {
345 /* Configure SPI to generate interrupt when transmit buffer is completely empty */
346 SPI1IMSKbits.SPITBEN = 1U;
347
348 /* ignore receive overflow for transmit only operation*/
349 SPI1CON1bits.IGNROV = 1U;
350
351 /* Enable transmit interrupt to complete the transfer in ISR context */
352 IEC2bits.SPI1TXIE = 1U;
353
354 }
355 else
356 {
357 /* Enable error interrupt for SRMT(last byte transfer in shift register)*/
358 SPI1IMSKbits.SRMTEN = 1U;
359 IEC2bits.SPI1EIE = 1U;
360 }
361
362 }
363 }
364
365 return isRequestAccepted;
366}

368void __attribute__((used)) SPI1RX_InterruptHandler (void)
369{
370 uint32_t nRxPending = 0;
371 uint32_t receivedData = 0;
372
373 /* Check Receive Buffer Element Count for watermark interrupt */
374 if ((SPI1STAT & _SPI1STAT_RXELM_MASK) != 0U)
375 {
376 size_t txCount = spi1Obj.txCount;
377 size_t rxCount = spi1Obj.rxCount;
378 while(SPI1STATbits.SPIRBE == false)
379 {
380 if (rxCount < spi1Obj.rxSize)
381 {
382 /* Receive buffer is not empty. Read the received data. */
383 receivedData = SPI1BUF;
384
385 ((uint8_t*)spi1Obj.rxBuffer)[rxCount] = (uint8_t)receivedData;
386 rxCount++;
387
388 spi1Obj.rxCount = rxCount;
389
390 if (rxCount == spi1Obj.rxSize)
391 {
392 if (txCount < spi1Obj.txSize)
393 {
394 /* Reception of all bytes is complete. However, there are few more
395 * bytes to be transmitted as txCount != txSize. Finish the
396 * transmission of the remaining bytes from the transmit interrupt. */
397
398 /* Disable the receive interrupt */
399 IEC1bits.SPI1RXIE = 0U;
400
401 /* Generate TX interrupt when buffer is completely empty */
402 SPI1IMSKbits.SPITBEN = 1U;
403
404 /* Enable the transmit interrupt. Callback will be given from the
405 * transmit interrupt, when all bytes are shifted out. */
406 IEC2bits.SPI1TXIE = 1U;
407 }
408 }
409 }
410 }
411 if (rxCount < spi1Obj.rxSize)
412 {
413 /* More bytes pending to be received .. */
414 SPI_FIFO_Fill();
415
416 nRxPending = spi1Obj.rxSize - rxCount;
417
418 if(nRxPending <= SPI1_FIFO_SIZE)
419 {
420 SPI1IMSKbits.RXMSK = (uint8_t)nRxPending;
421 }
422 else
423 {
424 SPI1IMSKbits.RXMSK = SPI1_FIFO_SIZE;
425 }
426 }
427 else
428 {
429 if(rxCount == spi1Obj.rxSize)
430 {
431 if (txCount == spi1Obj.txSize)
432 {
433 /* Clear receiver overflow error if any */
434 SPI1STATbits.SPIROV = 0U;
435
436 /* Disable receive interrupt */
437 IEC1bits.SPI1RXIE = 0U;
438
439 /* Transfer complete. Give a callback */
440 spi1Obj.transferIsBusy = false;
441
442 if(spi1Obj.callback != NULL)
443 {
444 uintptr_t context = spi1Obj.context;
445 spi1Obj.callback(context);
446 }
447 }
448 }
449 }
450 }
451
452 /* Clear SPI1 RX Interrupt flag */
453 /* This flag should cleared only after reading buffer */
454 IFS1bits.SPI1RXIF = 0U;
455}

484void __attribute__((used)) SPI1TX_InterruptHandler (void)
485{
486 /* If there are more words to be transmitted, then transmit them here and keep track of the count */
487 if((SPI1STAT & _SPI1STAT_SPITBE_MASK) == _SPI1STAT_SPITBE_MASK)
488 {
489 size_t txCount = spi1Obj.txCount;
490
491 while (txCount < spi1Obj.txSize)
492 {
493 if(SPI1STATbits.SPITBF != 0U)
494 {
495 break; /* Exit loop if buffer is full */
496 }
497 SPI1BUF = ((uint8_t*)spi1Obj.txBuffer)[txCount];
498 txCount++;
499
500 spi1Obj.txCount = txCount;
501 if (txCount == spi1Obj.txSize)
502 {
503 /* All bytes are submitted to the SPI module. Now, enable transmit
504 * interrupt for the shift register to empty */
505 SPI1IMSKbits.SPITBEN = 0U;
506 SPI1IMSKbits.SRMTEN = 1U;
507
508 /* Enable the error interrupt and disable the transmit interrupt*/
509 IEC2bits.SPI1EIE = 1U;
510 IEC2bits.SPI1TXIE = 0U;
511 }
512
513 }
514 }
515 /* Clear the transmit interrupt flag */
516 IFS2bits.SPI1TXIF = 0U;
517}
