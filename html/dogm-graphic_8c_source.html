54/******************************************************************************
55 * Changes the internal cursor by s pages
56 * s - number of pages to move
57 */
58uint8_t lcd_inc_page(int8_t s)
59{
60 uint8_t p = lcd_current_page;
61 p += s;
62 p %= LCD_RAM_PAGES; //all lcd have lcd_ram_pages which is power of two
63 lcd_current_page = p;
64 return p;
65}
66
67/******************************************************************************
68 * Changes the internal cursor by s columns, including wrapping (if selected)
69 * s - number of columns to move
70 */
71uint8_t lcd_inc_column(int16_t s)
72{
73 uint16_t c = lcd_current_column;
74 c += s;
75#if LCD_WRAP_AROUND == 1
76 while (c >= LCD_WIDTH) {
77 if (s > 0) lcd_inc_page(1);
78 else lcd_inc_page(-1);
79 if (s > 0) c -= LCD_WIDTH;
80 else c += LCD_WIDTH;
81 }
82#endif
83 lcd_current_column = c;
84 return c;
85}

159/******************************************************************************
160 * This function draws a bitmap from the current position on the screen.
161 * Parameters:
162 * progmem_image - prog_uint8_t array of columns aka the bitmap image
163 * pages - height of image in pages
164 * columns - width of image in pixels (or columns)
165 * style - Bit2: sets inverse mode
166 */
167void lcd_draw_image_P(uint8_t * progmem_image, uint8_t pages, uint8_t columns, uint8_t style)
168{
169 uint8_t i, j = 0;
170 uint8_t inv = (style & INVERT_BIT);
171 while (j < pages && (lcd_get_position_page() < LCD_RAM_PAGES)) {
172 for (i = 0; i < columns && (lcd_get_position_column() < LCD_WIDTH); i++) {
173 uint8_t tmp = *progmem_image++;
174 if (!inv)
175 lcd_data(tmp);
176 else
177 lcd_data(~tmp);
178 }
179 if (++j != pages && lcd_get_position_column() != 0)
180 lcd_move_xy(1, -columns);
181 }
182}

184/******************************************************************************
185 * This function draws a bitmap at any xy-position on the screen.
186 * Be aware that some pixels are deleted due to memory organization!
187 * Parameters:
188 * progmem_image - prog_uint8_t array of columns aka the bitmap image
189 * x - x start coordinate on the screen (in pixel)
190 * y - y start coordinate on the screen (in pixel)
191 * pages - height of image in pages
192 * columns - width of image in pixels
193 * style - Bit2: sets inverse mode
194 */
195void lcd_draw_image_xy_P(uint8_t * progmem_image, uint8_t x, uint8_t y, uint8_t pages, uint8_t columns, uint8_t style)
196{
197 uint16_t i, j;
198 uint8_t data = 0;
199 uint8_t inv = style & INVERT_BIT;
200 uint8_t offset = y & 0x7; //Optimized modulo 8
201 //If there is an offset, we must use an additional page
202 if (offset)
203 pages++;
204 //If there is not enough vertical space -> cut image
205 if (pages > LCD_RAM_PAGES - lcd_get_position_page())
206 pages = LCD_RAM_PAGES - lcd_get_position_page();
207 //Goto starting point and draw
208 lcd_moveto_xy((y >> 3), x);
209 for (j = 0; j < pages; j++) {
210 for (i = 0; i < columns && (lcd_get_position_column() < LCD_WIDTH); i++) {
211 data = 0;
212 if (!offset || j + 1 != pages)
213 data = (*progmem_image + j * columns + i) << offset;
214 if (j > 0 && offset)
215 data |= (*progmem_image + (j - 1) * columns + i) >> (8 - offset);
216 if (inv) lcd_data(~data);
217 else lcd_data(data);
218 }
219 if (j + 1 != pages)
220 lcd_move_xy(1, -columns);
221 }
222}

225/******************************************************************************
226 * This function clears an area of the screen
227 * pages - height of area in pages
228 * columns - width of area in pixels
229 * style - Bit2: sets inverse mode
230 * Cursor is moved to start of area after clear
231 */
232void lcd_clear_area(uint8_t pages, uint8_t columns, uint8_t style)
233{
234 uint8_t i, j, max;
235 uint8_t inv = (style & INVERT_BIT) ? 0xFF : 0;
236
237 if (pages > (max = LCD_RAM_PAGES - lcd_get_position_page())) {
238 pages = max;
239 }
240 if (columns > (max = LCD_WIDTH - lcd_get_position_column())) {
241 columns = max;
242 }
243
244 for (j = 0; j < pages; j++) {
245 for (i = 0; i < columns; i++) {
246 lcd_data(inv);
247 }
248 lcd_move_xy(1, -columns);
249 }
250 lcd_move_xy(-pages, 0);
251}
