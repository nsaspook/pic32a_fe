59void SCCP2_TimerInitialize(void)
60{
61 /* Disable Timer */
62 CCP2CON1 &= ~_CCP2CON1_ON_MASK;
63
64 CCP2CON1 = 0x20;
65
66 CCP2CON2 = 0x0;
67
68 CCP2CON3 = 0x0;
69
70 /* Clear counter */
71 CCP2TMR = 0x0;
72
73 /*Set period */
74 CCP2PR = 1999U;
75
76 IEC1 |= _IEC1_CCT2IE_MASK;
77
78}

113void __attribute__((used)) CCT2_InterruptHandler (void)
114{
115 /* Additional local variable to prevent MISRA C violations (Rule 13.x) */
116 uintptr_t context = sccp2Obj.context;
117 uint32_t status = IFS1bits.CCT2IF;
118 IFS1 &= ~_IFS1_CCT2IF_MASK;
119
120 if((sccp2Obj.callback_fn != NULL))
121 {
122 sccp2Obj.callback_fn(status, context);
123 }
124}
